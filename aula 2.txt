#include <stdio.h>

int main() {

    float salario, reajuste, novo_salario;
    int percentual;

    printf("Digite o salário:\n");
    scanf("%f", &salario);

    if (salario <= 400.00) {
        percentual = 15;
    }
    else if (salario <= 800.00) {
        percentual = 12;
    }
    else if (salario <= 1200.00) {
        percentual = 10;
    }
    else if (salario <= 2000.00) {
        percentual = 7;
    }
    else {
        percentual = 4;
    }

    reajuste = salario * percentual / 100;
    novo_salario = salario + reajuste;

    printf("Novo salario = %.2f\n", novo_salario);
    printf("Valor do reajuste = %.2f\n", reajuste);
    printf("Indice do reajuste = %d\n", percentual);

    return 0;
}

#include <stdio.h>

int main() {
    char tipo;
    float litros, preco, desconto, valor;

    printf("Digite o tipo de combustivel (G para Gasolina ou A para Alcool):");
    scanf(" %c", &tipo);

    printf("Digite a quantidade de litros:");
    scanf("%f", &litros);

    if (tipo == 'G' || tipo == 'g') {
        preco = 4.39;

        if (litros <= 20) {
            desconto = 0.04;
        } else {
            desconto = 0.06;
        }
    }
    else if (tipo == 'A' || tipo == 'a') {
        preco = 3.19;

        if (litros <= 20) {
            desconto = 0.03;
        } else {
            desconto = 0.05;
        }
    }
    else {
        printf("Tipo de combustivel invalido!\n");
        return 1;
    }

    valor = litros * preco * (1 - desconto);

    printf("Valor com desconto de %.0f%%: %.2f\n", desconto * 100, valor);

    return 0;
}

#include <stdio.h>

        contador++;

    if (resposta == 's') {
        contador++;

    scanf(" %c", &resposta);

    if (resposta == 's') {
        contador++;
    }

    printf("Devia para a vitima?\n");
    scanf(" %c", &resposta);

    if (resposta == 's') {
        contador++;
    }

    printf("Ja trabalhou com a vitima?\n");
    scanf(" %c", &resposta);

    if (resposta == 's') {
        contador++;
    }

    if (contador == 1) {
        printf("INOCENTE!");
    }
    else if (contador == 2 ) {

    }

    return 0;
}
